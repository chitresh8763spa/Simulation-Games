<script>

    let state = {

        budget: 10.0,

        accuracy: 48, // Report avg [cite: 23]

        otif: 52,     // Report avg [cite: 23]

        invDays: 52,  // Report avg [cite: 24]

        hasCloud: false,

        hasAI: false,

        quarter: 1

    };


    function updateUI() {

        document.getElementById('budget').innerText = state.budget.toFixed(1);

        document.getElementById('accuracy').innerText = state.accuracy;

        document.getElementById('otif').innerText = state.otif;

        document.getElementById('inv').innerText = state.invDays;

        document.getElementById('btn-cloud').disabled = state.hasCloud || state.budget < 2;

        document.getElementById('btn-ai').disabled = state.hasAI || state.budget < 1.5;

    }


    function log(msg) {

        const lb = document.getElementById('log-box');

        lb.innerHTML = `<div class="log-entry">${msg}</div>` + lb.innerHTML;

    }


    function investCloud() {

        state.budget -= 2.0;

        state.hasCloud = true;

        state.otif += 5; 

        log("<span class='impact-green'>Cloud Migration:</span> Scalability improved. 39% more access to innovation. [cite: 129]");

        updateUI();

    }


    function investAI() {

        state.budget -= 1.5;

        state.hasAI = true;

        state.accuracy += 10;

        state.invDays -= 5;

        log("<span class='impact-green'>GenAI Deployment:</span> Forecast accuracy improved. (Report: AI boosts planning by 32%). [cite: 54]");

        updateUI();

    }


    function optimizeNetwork() {

        state.budget -= 0.5;

        log("<span class='impact-green'>Network Optimization:</span> Re-balancing flows to mitigate potential tariff impacts. [cite: 201]");

    }


    function nextQuarter() {

        state.quarter++;

        // Random Disruption based on Report Page 36

        if (Math.random() > 0.6) {

            const loss = state.hasCloud ? 0.6 : 1.0;

            state.budget -= loss;

            log(`<span class='impact-red'>Disruption:</span> Global Supply Shock! Loss: $${loss}M. (Legacy systems feel full $1.0M impact). [cite: 200]`);

        } else {

            log(`Quarter ${state.quarter}: Standard operations. Inventory holding costs apply.`);

        }

        updateUI();

    }

</script>
